API Kullanımı

// Canvas tabanlı büyüteç (v1.1 default — DOM kirletmez)
const mag = new Magnifier({
  shape: 'circle',
  zoom: 2,
  renderMode: 'canvas',   // SVG foreignObject → canvas snapshot
  autoStart: true,
});

// Runtime'da seçenekleri değiştir
mag.setOptions({ shape: 'rectangle', width: 360, height: 220, zoom: 3 });

// Haber sitesi: reklam alanlarında büyüteci gizle
new Magnifier({
  excludeSelectors: ['.ad-slot', '.gpt-ad', '[data-ad]'],
  keyboardShortcut: 'm',
  shortcutWithAlt: true,
});

// Tamamen kaldır
mag.destroy();
